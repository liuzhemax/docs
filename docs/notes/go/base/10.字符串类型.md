Go语言中的字符串是一个只读的byte类型的切片。

Go 字符串是 UTF-8 编码的文本。

示例一：
```go
package main

import "fmt"

func main() {
    // 创建一个字符串
    str := "Hello,世界"

    // 遍历打印字符串中的每个字符
    for index, runeValue := range str {
        fmt.Printf("%#U starts at byte position %d\n", runeValue, index)
    }
}

```

示例二：
```go
package main

import (
    "fmt"
    "strings"
    "time"
)

func main() {
    // 最普通的拼接字符串
    str0 := "hello"
    str1 := "world"
    str2 := "!"
    showStr := str0 + " " + str1 + " " + str2
    fmt.Println(showStr)

    // 使用strings.Builder 优化字符串拼接
    var str strings.Builder
    str.WriteString(str0)
    str.WriteRune(' ')
    str.WriteString(str1)
    str.WriteRune(' ')
    str.WriteString(str2)
    fmt.Println(str.String())

    // 字符串拼接性能测试
    start := time.Now()
    for i := 0; i < 100000; i++ {
        showStr = showStr + "0"
        // str.WriteString("0")
    }
    fmt.Println(showStr)
    // fmt.Println(str.String())
    fmt.Println("Time elapsed:", time.Since(start))
}

```
