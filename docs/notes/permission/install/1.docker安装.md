### 环境准备

```bash
#安装docker
wget https://mirrors.aliyun.com/docker-ce/linux/centos/docker-ce.repo -O /etc/yum.repos.d/docker-ce.repo
yum -y install docker-ce
systemctl enable docker && systemctl start docker 

#配置加速
cat > /etc/docker/daemon.json << EOF
{
  "registry-mirrors": ["https://docker.m.daocloud.io","https://docker.chenby.cn","http://dockerhub.aliangedu.cn"]
}
EOF
#重启服务
systemctl restart docker
#查看
docker info
```

### 拉取代码

```bash
yum -y install git 
cd /opt/
git clone https://gitee.com/lucky_liuzhe/permission_api.git
git clone https://gitee.com/lucky_liuzhe/permission_web.git
```

### 环境配置

```bash
#前端配置
cd permission_web/
#修改VUE_APP_API_HOST值为自己服务器地址
vim .env.production
VUE_APP_API_HOST=192.168.0.11

#后端配置
cd ../permission_api/
#修改数据库地址为服务器地址，用户名密码及端口默认即可
vim devops_api/settings.py
DATABASES = {
    'default': {
        'ENGINE': 'django.db.backends.mysql',
        'NAME': 'permission',
        'USER': 'root',
        'PASSWORD': '123456',
        'HOST': '192.168.0.11',
        'PORT': '3306',
    }
}
```

### 部署安装

```bash
#添加权限
chmod +x install.sh
#执行安装（安装需要一定的时间，请耐心等待）
bash install.sh 
```
